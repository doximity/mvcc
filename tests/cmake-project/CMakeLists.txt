# A CMake CUDA project built with the mvcc toolkit (tests/run.sh).
cmake_minimum_required(VERSION 3.18)
project(mvcc_cmake_test LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(CUDAToolkit REQUIRED)
message(STATUS "CUDAToolkit ${CUDAToolkit_VERSION} at ${CUDAToolkit_LIBRARY_DIR}; archs=${CMAKE_CUDA_ARCHITECTURES}")
add_library(kern STATIC kern.cu)
set_target_properties(kern PROPERTIES CUDA_SEPARABLE_COMPILATION ON POSITION_INDEPENDENT_CODE ON)
target_compile_options(kern PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr> $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-Wno-deprecated-declarations>)
target_link_libraries(kern PRIVATE CUDA::cudart)
add_executable(app main.cpp)
target_link_libraries(app PRIVATE kern CUDA::cudart)
target_include_directories(app PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
